set(YORADIO_OPUS_BACKEND "c" CACHE STRING "Opus implementation: c, gcc-asm, optimized-asm, hoisted-asm")
set_property(CACHE YORADIO_OPUS_BACKEND PROPERTY STRINGS c gcc-asm optimized-asm hoisted-asm)
if(NOT YORADIO_OPUS_BACKEND MATCHES "^(c|gcc-asm|optimized-asm|hoisted-asm)$")
  message(FATAL_ERROR "Unknown YORADIO_OPUS_BACKEND")
endif()
if(NOT CONFIG_YORADIO_OGG_OPUS AND NOT YORADIO_OPUS_BACKEND STREQUAL "c")
  message(FATAL_ERROR "Opus ASM backend requires CONFIG_YORADIO_OGG_OPUS")
endif()
if(CONFIG_YORADIO_OGG_OPUS)
  option(YORADIO_OPUS_LOW_RAM "Diagnostic SILK loan, compact autocorrelation and PLC IRAM" OFF)
  if(YORADIO_OPUS_LOW_RAM AND NOT YORADIO_ESP8266_DIAGNOSTIC)
    message(FATAL_ERROR "Low-RAM Opus requires diagnostic firmware until device qualification")
  endif()
  option(YORADIO_OPUS_PCM_LEASES "Experimental per-frame PCM ownership API" OFF)
  if(YORADIO_OPUS_PCM_LEASES AND NOT YORADIO_ESP8266_DIAGNOSTIC)
    message(FATAL_ERROR "PCM leases require diagnostic firmware until queue qualification")
  endif()
  option(YORADIO_OPUS_WORD_ASM "A/B: private Opus LX106 word accesses without volatile-C MEMW" OFF)
  option(YORADIO_OPUS_ICDF_FLASH_WORD "A/B: decode static flash ICDF tables with word loads" OFF)
  option(YORADIO_OPUS_FIR_FLASH_WORD "A/B: SILK resampler flash coefficients as word pairs" OFF)
  option(YORADIO_OPUS_CELT_DECODE_ONLY "A/B: specialize CELT bands for decoding" OFF)
  option(YORADIO_OPUS_ROTATION_LX106 "A/B: exact adjacent CELT rotation in LX106 assembly" OFF)
  option(YORADIO_OPUS_DIV_ONCE "A/B: evaluate CELT division reciprocal once" OFF)
  set(YORADIO_OPUS_PROFILE_STAGE 0 CACHE STRING "Diagnostic selected Opus stage, 0 off / 1..11")
  if(NOT YORADIO_OPUS_PROFILE_STAGE MATCHES "^([0-9]|10|11)$")
    message(FATAL_ERROR "Opus profile stage must be 0..11")
  endif()
  if(YORADIO_OPUS_PROFILE_STAGE GREATER 0 AND
     (NOT YORADIO_ESP8266_DIAGNOSTIC OR NOT YORADIO_ESP8266_OPUS_BENCHMARK OR YORADIO_ESP8266_OPUS_BENCHMARK_OUTPUT))
    message(FATAL_ERROR "Opus stage profile requires diagnostic raw-only benchmark")
  endif()
  if(YORADIO_OPUS_PROFILE_STAGE GREATER 0 AND NOT CONFIG_ESP8266_DEFAULT_CPU_FREQ_160)
    message(FATAL_ERROR "Opus stage experiment requires the matched fixed CPU160 benchmark profile")
  endif()
  file(GLOB OPUS_SOURCES upstream/src/*.c upstream/celt/*.c
       upstream/silk/*.c upstream/silk/fixed/*.c)
  if(NOT YORADIO_OPUS_BACKEND STREQUAL "c")
    include(asm/select.cmake)
  endif()
  idf_component_register(SRCS ${OPUS_SOURCES}
      opus_memory.c opus_stage_profile.c opus_rotation_lx106.S ogg_opus_demux.c native_opus.c
      INCLUDE_DIRS . upstream/include
      PRIV_INCLUDE_DIRS upstream/celt upstream/silk upstream/silk/fixed upstream/src)
  target_compile_definitions(${COMPONENT_LIB} PRIVATE YORADIO_OPUS_BOUNDED=1)
  if(YORADIO_OPUS_LOW_RAM)
    # scratch-mark ABI is shared with any consumer of opus_memory.h.
    target_compile_definitions(${COMPONENT_LIB} PUBLIC YORADIO_OPUS_CELT_SILK_SCRATCH=1)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE YORADIO_OPUS_AUTOCORR_COMPACT=1 YORADIO_OPUS_SILK_PLC_IRAM=1)
  endif()
  if(YORADIO_OPUS_PCM_LEASES)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC YORADIO_OPUS_PCM_LEASES=1)
  endif()
  target_compile_definitions(${COMPONENT_LIB} PUBLIC YORADIO_OPUS_PROFILE_STAGE=${YORADIO_OPUS_PROFILE_STAGE})
  if(YORADIO_OPUS_WORD_ASM)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE YORADIO_OPUS_WORD_ASM=1)
  endif()
  if(YORADIO_OPUS_ICDF_FLASH_WORD)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE YORADIO_OPUS_ICDF_FLASH_WORD=1)
  endif()
  if(YORADIO_OPUS_FIR_FLASH_WORD)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE YORADIO_OPUS_FIR_FLASH_WORD=1)
  endif()
  if(YORADIO_OPUS_CELT_DECODE_ONLY)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE YORADIO_OPUS_CELT_DECODE_ONLY=1)
  endif()
  if(YORADIO_OPUS_ROTATION_LX106)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE YORADIO_OPUS_ROTATION_LX106=1)
  endif()
  if(YORADIO_OPUS_DIV_ONCE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE YORADIO_OPUS_DIV_ONCE=1)
  endif()
  target_compile_options(${COMPONENT_LIB} PRIVATE -O3 -fwrapv -fstack-usage)
  # The SDK sets -mlongcalls for C/C++, but not ASM. Preserve the GCC export
  # assembler relaxation for calls from flash to ROM/IRAM and libgcc.
  if(NOT YORADIO_OPUS_BACKEND STREQUAL "c")
    target_compile_options(${COMPONENT_LIB} PRIVATE "$<$<COMPILE_LANGUAGE:ASM>:-mlongcalls>" "$<$<COMPILE_LANGUAGE:ASM>:-g0>")
  endif()
else()
  idf_component_register()
endif()
