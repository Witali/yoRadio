cmake_minimum_required(VERSION 3.16)
project(small_tv_color_contract C)
add_library(color_contract SHARED bridge.c ../../app/media/color_convert.c)
target_include_directories(color_contract PRIVATE ../../app/include)
target_compile_features(color_contract PRIVATE c_std_11)
if(MSVC)
  target_compile_options(color_contract PRIVATE /W4 /WX)
else()
  target_compile_options(color_contract PRIVATE -Wall -Wextra -Werror)
endif()
