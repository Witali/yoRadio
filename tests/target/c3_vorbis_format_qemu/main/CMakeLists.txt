idf_component_register(SRCS "main.c" INCLUDE_DIRS "."
    REQUIRES esp_audio_codec freertos heap esp_system esp_rom)
if(CMAKE_BUILD_EARLY_EXPANSION)
    return()
endif()
if(NOT IDF_TARGET STREQUAL "esp32c3" OR NOT CONFIG_IDF_TARGET_ESP32C3)
    message(FATAL_ERROR "This public raw format fixture requires ESP32-C3 RISC-V")
endif()
if(NOT "${IDF_VERSION_MAJOR}" STREQUAL "6" OR
   NOT "${IDF_VERSION_MINOR}" STREQUAL "1" OR NOT "${IDF_VERSION_PATCH}" STREQUAL "0")
    message(FATAL_ERROR "This fixture requires pinned SDK 6.1.0")
endif()
if(NOT DEFINED FORMAT_EXPECT_LOW_RATE_REJECTION OR
   NOT "${FORMAT_EXPECT_LOW_RATE_REJECTION}" MATCHES "^[01]$")
    message(FATAL_ERROR "Explicit FORMAT_EXPECT_LOW_RATE_REJECTION=1 baseline or0 acceptance required")
endif()
set(M "$ENV{C3_VORBIS_CANDIDATE}")
if(M STREQUAL "")
    message(FATAL_ERROR "Explicit source-pinned C3 variant directory required")
endif()
# The campaign must bind its separately reviewed frozen setup.c bytes.
string(LENGTH "${FORMAT_SETUP_SHA256}" setup_sha_length)
if(NOT setup_sha_length EQUAL 64 OR NOT "${FORMAT_SETUP_SHA256}" MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Explicit reviewed FORMAT_SETUP_SHA256 is required")
endif()
set(pinned_setup "${FORMAT_SETUP_SHA256}")
set(pinned_output "7e7a49a1ba68b2417514ab4b0adb37c9ed31e2d1b65032fa06dbf4b2830c25e2")
set(pinned_abi "dd26a294cf3887853e12174f3958ea9cccdaa9425ff6a7b21d46faa350bde61c")
foreach(name setup output abi)
    if(name STREQUAL "abi")
        set(path "${M}/vorbis_repair_abi.h")
    else()
        set(path "${M}/vorbis_repair/${name}.c")
    endif()
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Missing pinned C3 source: ${name}")
    endif()
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL "${pinned_${name}}")
        message(FATAL_ERROR "C3 source changed: ${name}; review new source pins separately")
    endif()
endforeach()
foreach(name codec simple_dec)
    set(path "$ENV{YORADIO_AUDIO_CODEC_COMPONENT}/lib/esp32c3/libesp_audio_${name}.a")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Missing pinned ESP32-C3 codec archive")
    endif()
    file(SHA256 "${path}" actual)
    if(name STREQUAL "codec")
        set(expected "311caa814095b098e476b28e46d55623ef70c26b0d73d69ba5832e154ec8d909")
    else()
        set(expected "232650ec81ae5e10ecbe1b2c86de2d1adf8f1b793f5ef343241c6e510db3e71f")
    endif()
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "ESP32-C3 codec archive identity changed")
    endif()
endforeach()
target_compile_definitions(${COMPONENT_LIB} PRIVATE
    FORMAT_EXPECT_LOW_RATE_REJECTION=${FORMAT_EXPECT_LOW_RATE_REJECTION})
target_sources(${COMPONENT_LIB} PRIVATE
    "${M}/vorbis_repair/setup.c" "${M}/vorbis_repair/output.c")
target_link_options(${COMPONENT_LIB} INTERFACE
    "-Wl,--wrap=esp_vorbis_dec_open" "-Wl,--wrap=esp_vorbis_dec_decode")
