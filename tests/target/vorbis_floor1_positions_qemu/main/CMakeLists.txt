set(test_source "main.c")
if(FLOOR1_RENDER_MODE STREQUAL "on")
    set(test_source "render.c")
endif()
idf_component_register(SRCS "${test_source}" INCLUDE_DIRS "."
    REQUIRES esp_audio_codec freertos heap esp_system esp_rom)
if(CMAKE_BUILD_EARLY_EXPANSION)
    return()
endif()
if(IDF_TARGET STREQUAL "esp32" AND CONFIG_IDF_TARGET_ESP32)
    set(expected_sdk "6.0.2")
    set(expected_codec "92ba997358f50994d41422438c78bf2d7ee86dc1efc265f5356a66e221ae2e58")
elseif(IDF_TARGET STREQUAL "esp32c3" AND CONFIG_IDF_TARGET_ESP32C3)
    set(expected_sdk "6.1.0")
    set(expected_codec "311caa814095b098e476b28e46d55623ef70c26b0d73d69ba5832e154ec8d909")
else()
    message(FATAL_ERROR "Select the ESP32 or ESP32-C3 standalone raw test")
endif()
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" STREQUAL expected_sdk)
    message(FATAL_ERROR "Wrong pinned SDK for selected target")
endif()
set(variant "$ENV{VORBIS_FLOOR1_MAIN}")
foreach(part setup output abi floor)
    if(part STREQUAL "abi")
        set(source "${variant}/vorbis_repair_abi.h")
    elseif(part STREQUAL "floor")
        set(source "${variant}/vorbis_repair/floor1_render.h")
    else()
        set(source "${variant}/vorbis_repair/${part}.c")
    endif()
    file(SHA256 "${source}" actual)
    if(NOT actual STREQUAL "${FLOOR1_${part}_SHA256}")
        message(FATAL_ERROR "Unreviewed ${part} source")
    endif()
endforeach()
file(SHA256 "$ENV{YORADIO_AUDIO_CODEC_COMPONENT}/lib/${IDF_TARGET}/libesp_audio_codec.a" actual)
if(NOT actual STREQUAL expected_codec)
    message(FATAL_ERROR "Wrong public codec archive")
endif()
target_sources(${COMPONENT_LIB} PRIVATE
    "${variant}/vorbis_repair/setup.c" "${variant}/vorbis_repair/output.c")
target_link_options(${COMPONENT_LIB} INTERFACE
    "-Wl,--wrap=esp_vorbis_dec_open" "-Wl,--wrap=esp_vorbis_dec_decode")
if(FLOOR1_RENDER_MODE STREQUAL "on")
    target_link_options(${COMPONENT_LIB} INTERFACE "-Wl,--wrap=floor1_inverse2")
elseif(NOT FLOOR1_RENDER_MODE STREQUAL "off")
    message(FATAL_ERROR "Explicit FLOOR1_RENDER_MODE on/off is required")
endif()
