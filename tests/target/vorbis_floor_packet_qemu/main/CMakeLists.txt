idf_component_register(SRCS "main.c" INCLUDE_DIRS "."
                      REQUIRES freertos esp_system native_vorbis_numeric)
if(CMAKE_BUILD_EARLY_EXPANSION)
    return()
endif()
if(IDF_TARGET STREQUAL "esp32" AND CONFIG_IDF_TARGET_ESP32)
    set(expected_sdk "6.0.2")
elseif(IDF_TARGET STREQUAL "esp32c3" AND CONFIG_IDF_TARGET_ESP32C3)
    set(expected_sdk "6.1.0")
else()
    message(FATAL_ERROR "Only isolated ESP32/C3 floor packet tests are supported")
endif()
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" STREQUAL expected_sdk)
    message(FATAL_ERROR "Wrong pinned SDK version")
endif()
foreach(part source header quant_source quant_header codebook_source codebook_header setup_source setup_header floor_packet_source floor_packet_header component)
    if(part STREQUAL "source")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/floor0.c")
    elseif(part STREQUAL "header")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/include/native_vorbis_floor0.h")
    elseif(part STREQUAL "quant_source")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/quant.c")
    elseif(part STREQUAL "quant_header")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/include/native_vorbis_quant.h")
    elseif(part STREQUAL "codebook_source")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/codebook.c")
    elseif(part STREQUAL "codebook_header")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/include/native_vorbis_codebook.h")
    elseif(part STREQUAL "setup_source")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/setup.c")
    elseif(part STREQUAL "setup_header")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/include/native_vorbis_setup.h")
    elseif(part STREQUAL "floor_packet_source")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/floor_packet.c")
    elseif(part STREQUAL "floor_packet_header")
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/include/native_vorbis_floor_packet.h")
    else()
        set(file "$ENV{FLOOR_PACKET_COMPONENT}/CMakeLists.txt")
    endif()
    file(SHA256 "${file}" actual)
    if(NOT actual STREQUAL "${FLOOR0_${part}_SHA256}")
        message(FATAL_ERROR "Frozen numeric ${part} hash mismatch")
    endif()
endforeach()
target_compile_definitions(${COMPONENT_LIB} PRIVATE
                          FLOOR_PACKET_CASES_SHA256="${FLOOR0_CASES_SHA256}")
target_compile_options(${COMPONENT_LIB} PRIVATE -fno-fast-math)
target_link_libraries(${COMPONENT_LIB} PRIVATE m)
# This flag belongs only to the isolated test, not the production component.
idf_component_get_property(numeric_library native_vorbis_numeric COMPONENT_LIB)
target_compile_options(${numeric_library} PRIVATE -fstack-usage)
