def require(value, message):
    if not value:
        raise RuntimeError(message)


def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def write(path, value):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, ensure_ascii=True)+'\n', encoding='utf-8')


def ident(path):
    path = Path(path).resolve(strict=True)
    before = path.stat()
    with path.open('rb') as stream:
        digest = hashlib.file_digest(stream, 'sha256').hexdigest()
    after = path.stat()
    require((before.st_size, before.st_mtime_ns, before.st_ino) ==
            (after.st_size, after.st_mtime_ns, after.st_ino), 'Input changed while hashed')
    return dict(path=str(path), bytes=after.st_size, sha256=digest)


def same(binding):
    require(ident(binding['path']) == binding, 'Frozen input changed: '+binding['path'])

def git(cwd, *args, data=None):

def module(path):

def release(path):
    result = read(path)
    require(result.get('schema') == 'cyd-final-production-release-v1' and result.get('released') is True,
            'Root has not explicitly released FINAL_COMMIT')
    for key in ('final_commit', 'task_profile_reference_commit'):
        require(re.fullmatch('[0-9a-f]{40}', str(result.get(key))), 'Explicit full commit required: '+key)
    for key in ('source_release_id', 'campaign_id'):
        require(re.fullmatch('[a-zA-Z0-9_-]{3,80}', str(result.get(key))), 'Explicit release/campaign identity required')
    task_worktree = Path(result['worktree']).resolve()
    require(task_worktree.parent == REPO/'.worktree' and
            re.fullmatch(r'cyd-ogg-production-[0-9]{8}(?:-[A-Za-z0-9_-]+)?', task_worktree.name),
            'Production task must be a bounded repository-local worktree with a fresh date/suffix')
    require(result['branch'] == 'codex/'+task_worktree.name, 'Own matching codex task branch required')
    require(result.get('sparse_checkout_approved') is True, 'Explicit standard sparse-checkout review required')
    require(result['evidence_files'] and result.get('c3_host_baseline_report'), 'Final task evidence and donor baseline are required')
    for name in ('campaign.py', 'invoke-build.ps1', 'README.md'):
        require(result['reviewed_campaign_files_sha256'].get(name) == ident(HERE/name)['sha256'],
                'Root review does not bind exact prepared recipe: '+name)
    require(git(REPO, 'rev-parse', '--verify', result['final_commit']+'^{commit}').decode().strip() == result['final_commit'],
            'FINAL_COMMIT is not already local; no fetch is permitted')
    return result

def sparse_patterns(r):
    captures = set()
    for name in evidence_files(r):
        p = Path(name)
        require(not p.is_absolute() and '..' not in p.parts, 'Evidence paths must be committed repository-relative paths')
        if p.parts[:2] == ('tests', 'results'):
            require(len(p.parts) >= 4, 'Evidence must identify actual files in a bounded capture')
            captures.add(p.parts[2])
        require(not p.parts[0] == 'firmware', 'Old firmware is not an implicit final evidence input')
    return '/*\n!/tests/results/*\n!/firmware/*\n'+''.join('/tests/results/'+x+'/\n' for x in sorted(captures))


def disk_plan(r, patterns, directory):
    rows = []
    for record in git(REPO, 'ls-tree', '-r', '-l', '-z', r['final_commit']).split(b'\0'):
        if not record:
            continue
        meta, name = record.split(b'\t', 1)
        mode, kind, oid, size = meta.split()
        require(kind == b'blob', 'Submodules require a separate offline review')
        rows.append((os.fsdecode(name), int(size)))
    rules = directory/'sparse-patterns.txt'
    rules.write_text(patterns)
    chosen = set(os.fsdecode(x) for x in git(REPO, 'sparse-checkout', 'check-rules', '--no-cone',
                 '--rules-file', str(rules), '-z', data=b'\0'.join(os.fsencode(p) for p, s in rows)+b'\0').split(b'\0') if x)
    material = [(p, s) for p, s in rows if p in chosen]
    checkout = sum((s+4095)//4096*4096 for p, s in material)
    # Measured full GCC reference was 176,250,044 B. Budgets deliberately exceed
    # that reference; all six trees/configurations and original logs are retained.
    mib = 1024**2
    components = dict(sparse_checkout_rounded_4k=checkout, six_gcc_trees=6*256*mib,
                      six_clang_configurations=6*96*mib, two_c3_configurations_and_objects=2*256*mib,
                      donor_host_and_fixtures=128*mib, deduplicated_dependency_snapshots=512*mib,
                      original_logs_and_reports=256*mib, six_firmware_handoffs=64*mib,
                      operational_reserve=1024*mib)
    required = math.ceil(sum(components.values())/1024**3)*1024**3
    result = dict(components=components, minimum_free_before_launch=required,
                  selected_tracked_files=len(material), selected_git_blob_bytes=sum(s for p, s in material),
                  free_now=shutil.disk_usage(REPO).free, estimate_not_guarantee=True,
                  required_evidence_packages_included=sorted(chosen & set(evidence_files(r))))
    require(set(evidence_files(r)) <= chosen, 'Sparse plan omits required released evidence')
    write(directory/'disk-plan.json', result)
    require(result['free_now'] >= required, 'Insufficient measured free space for retained campaign; no automatic cleanup')
    return result


def process(argv, directory, label, record, cwd, environment=None, expected_exit=0):
    require(expected_exit == 0 or record['phase'] == 'wrapper-test' and label == 'shim-host-throw' and expected_exit == 1,
            'Nonzero outcomes are allowed only by the explicit host error-propagation control')
    stdout, stderr = directory/(label+'.stdout.log'), directory/(label+'.stderr.log')
    with stdout.open('xb') as out, stderr.open('xb') as err:
        child = subprocess.Popen(list(map(str, argv)), cwd=cwd, env=environment or env(), stdout=out, stderr=err)
        item = dict(argv=list(map(str, argv)), pid=child.pid, state='RUNNING', started_epoch=time.time(), expected_exit=expected_exit)
        record['children'].append(item)
        write(directory/'record.json', record)
        print(json.dumps(dict(phase=record['phase'], child_pid=child.pid, state='RUNNING', record=str(directory/'record.json'))), flush=True)
        code = child.wait()  # No observation timeout, restart, process kill or deadline extension.
    item.update(state='TERMINAL', exit_status=code, ended_epoch=time.time(), logs=[ident(stdout), ident(stderr)])
    write(directory/'record.json', record)
    require(code == expected_exit, 'Unexpected terminal result preserved; never restart or overwrite this phase')
    return stdout.read_text(encoding='utf-8', errors='replace')


def source_check(r, w):
    require(git(w, 'rev-parse', 'HEAD').decode().strip() == r['final_commit'], 'Final source commit changed')
    require(git(w, 'branch', '--show-current').decode().strip() == r['branch'], 'Wrong task branch')
    require(not git(w, 'diff', '--name-only', 'HEAD').strip(), 'Tracked source/index is dirty')
    api = module(w/'tools/export_cyd_production.py')
    state = api.source_snapshot(w)
    require(not state['source_dirty'] and not state['untracked'], 'Clean committed source required; AllowDirty is forbidden')
    allowed = [w/'firmware/development'/('esp32-cyd2usb-native-ogg-final-'+name) for name, b, c in ROWS]
    for name in git(w, 'ls-files', '--others', '--exclude-standard', '-z').split(b'\0'):
        if name:
            require(any((w/os.fsdecode(name)).is_relative_to(p) for p in allowed), 'Unrelated untracked material in isolated checkout')
    return state


def board_tasks(r, w):
    result = {}
    baseline = r['task_profile_reference_commit']
    for name in ('app_main.c', 'network_runtime.c', 'audio_service.c'):
        rel = (CYD/'main'/name).as_posix()
        before = git(w, 'show', baseline+':'+rel).decode().replace('\r\n', '\n')
        after = (w/rel).read_text()
        calls = lambda x: re.findall(r'\bxTaskCreatePinnedToCore\([^;]+;', x, re.S)
        require(calls(before) == calls(after) and calls(after), 'Original core/priority/stack task calls changed: '+name)
        result[name] = dict(calls=calls(after), source=ident(w/rel))
        if name == 'audio_service.c':
            pattern = r'static const pipeline_task_t task_config\[3\] = \{(.*?)\n    \};'
            require(re.search(pattern, before, re.S)[1] == re.search(pattern, after, re.S)[1], 'Audio task binding table changed')
            result[name]['pipeline_table'] = re.search(pattern, after, re.S)[1]
    rel = (CYD/'main/board_config.h').as_posix()
    require(git(w, 'show', baseline+':'+rel).replace(b'\r\n', b'\n') == (w/rel).read_bytes().replace(b'\r\n', b'\n'), 'CYD hardware/stack header changed')
    result['board_config'] = ident(w/rel)
    rel = (CYD/'partitions.csv').as_posix()
    require(git(w, 'show', baseline+':'+rel).replace(b'\r\n', b'\n') == (w/rel).read_bytes().replace(b'\r\n', b'\n'), 'CYD OTA/partition source layout changed')
    result['partition_source'] = ident(w/rel)
    return result

def configure_seed(config, custom=None, gcc_config=None):
    config.parent.mkdir(parents=True, exist_ok=False)
    if gcc_config:
        text = gcc_config.read_text()
        keys = r'(?:IDF_TOOLCHAIN(?:_CLANG|_GCC)?|LIBC_(?:PICOLIBC(?:_NEWLIB_COMPATIBILITY)?|NEWLIB(?:_NANO_FORMAT)?))'
        text = re.sub(r'(?m)^(?:CONFIG_'+keys+r'=.*|# CONFIG_'+keys+r' is not set)\r?\n?', '', text)
        text += '\nCONFIG_IDF_TOOLCHAIN="clang"\nCONFIG_LIBC_NEWLIB=y\nCONFIG_LIBC_NEWLIB_NANO_FORMAT=y\n'
    else:
        text = 'CONFIG_IDF_TOOLCHAIN="gcc"\n'
        if custom is not None:
            text += '\n'.join('CONFIG_'+key+'='+('y' if (key.endswith('CUSTOM') == custom) else 'n') for key in ('YORADIO_FLAC_DECODER_ESPRESSIF', 'YORADIO_FLAC_DECODER_CUSTOM'))+'\n'
            text += '\n'.join('# CONFIG_'+key+' is not set' for key in OFF)+'\n'
    config.write_text(text, encoding='utf-8')

def tokens(command):
    argc = ctypes.c_int()
    parse = ctypes.windll.shell32.CommandLineToArgvW
    parse.argtypes = [wintypes.LPCWSTR, ctypes.POINTER(ctypes.c_int)]
    parse.restype = ctypes.POINTER(wintypes.LPWSTR)
    values = parse(command, ctypes.byref(argc))
    require(bool(values), 'Windows argv parsing failed')
    try:
        return [values[i] for i in range(argc.value)]
    finally:
        free = ctypes.windll.kernel32.LocalFree
        free.argtypes = [ctypes.c_void_p]
        free(ctypes.cast(values, ctypes.c_void_p))


def entry_tokens(entry, response_bindings=None):
    argv = list(entry['arguments']) if 'arguments' in entry else tokens(entry['command'])
    def expand(items, depth=0):
        require(depth < 12, 'Response-file recursion')
        result = []
        for item in items:
            if item.startswith('@'):
                path = Path(item[1:]); path = path if path.is_absolute() else Path(entry['directory'])/path
                if response_bindings is not None:
                    response_bindings[str(path.resolve())] = ident(path)
                result.extend(expand(tokens('response '+path.read_text())[1:], depth+1))
            else:
                require(item not in ('&&', '|', '>', '<'), 'Shell command wrappers are unsupported')
                result.append(item)
        return result
    argv = expand(argv)
    require('clang' in Path(argv[0]).name or 'gcc' in Path(argv[0]).name or 'g++' in Path(argv[0]).name, 'Actual compiler driver required')
    require(tokens(subprocess.list2cmdline(argv)) == argv, 'Windows compile argument roundtrip failed')
    return argv


def input_path(entry):
    p = Path(entry['file'])
    return (p if p.is_absolute() else Path(entry['directory'])/p).resolve()


def object_path(entry):
    argv = entry_tokens(entry)
    require('-o' in argv, 'Actual compiler output path is required')
    p = Path(argv[argv.index('-o')+1])
    return (p if p.is_absolute() else Path(entry['directory'])/p).resolve()


def store_inputs(paths, run):
    result = []
    store = run/'dependency-bytes'
    store.mkdir(exist_ok=True)
    for path in sorted(set(Path(p).resolve() for p in paths)):
        binding = ident(path)
        target = store/binding['sha256']
        if not target.exists():
            with target.open('xb') as out, path.open('rb') as stream:
                shutil.copyfileobj(stream, out)
        require(ident(target)['sha256'] == binding['sha256'], 'Dependency byte-store corruption')
        same(binding)
        result.append(dict(**binding, snapshot=str(target)))
    return result


def gcc_closure(build, entries, dep, donor, directory, record, run, w):
    ninja = dep/('tools-v6.1' if donor else 'tools-v6.0.2')/'tools/ninja/1.12.1/ninja.exe'
    text = process([ninja, '-C', build, '-t', 'deps'], directory, 'ninja-deps', record, w)
    dependencies, current = {}, None
    for line in text.splitlines():
        if line and not line.startswith(' '):
            match = re.match(r'(.+): #deps .*\((VALID|STALE)\)', line)
            current = match[1] if match else None
            if current:
                require(match[2] == 'VALID', 'Stale Ninja consumed dependency record')
                dependencies[current.replace('\\', '/')] = []
        elif current and line.startswith('    '):
            p = Path(line[4:]); dependencies[current.replace('\\', '/')].append(p if p.is_absolute() else build/p)
    all_inputs, outputs, response = set(), [], {}
    for entry in entries:
        argv = entry_tokens(entry, response)
        obj = object_path(entry)
        key = obj.relative_to(build).as_posix()
        require(key in dependencies and obj.is_file(), 'Actual compiled object/dependency record missing: '+key)
        raw = obj.read_bytes()
        require(len(raw) >= 52 and raw[:6] == b'\x7fELF\x01\x01' and struct.unpack_from('<H', raw, 18)[0] == (243 if donor else 94), 'Actual object architecture differs')
        all_inputs.update(dependencies[key]); all_inputs.add(input_path(entry))
        outputs.append(dict(command=entry, actual_argv=argv, object=ident(obj), dependencies_key=key))
    all_inputs.update(Path(p) for p in response)
    return dict(actual_objects=outputs, inputs=store_inputs(all_inputs, run), responses=list(response.values()),
                ninja_dependencies=ident(directory/'ninja-deps.stdout.log'))


def frontend_args(entry):
    clean, argv, i = [], entry_tokens(entry), 0
    while i < len(argv):
        if argv[i] in ('-o', '-MF', '-MT', '-MQ'):
            i += 2
        elif argv[i] in ('-c', '-MD', '-MMD', '-MP'):
            i += 1
        else:
            clean.append(argv[i]); i += 1
    return clean


def make_dependencies(path, cwd):
    text = path.read_text().replace('\\\r\n', '').replace('\\\n', '')
    require(text.startswith('final_dependency:'), 'Unexpected actual dependency target')
    result = []
    for token in re.findall(r'(?:\\[ #:\\]|[^\s])+', text.split(':', 1)[1]):
        p = Path(re.sub(r'\\([ #:\\])', r'\1', token))
        result.append((p if p.is_absolute() else Path(cwd)/p).resolve(strict=True))
    return result


def lint_selection(entries, w, build):
    result = []
    for entry in entries:
        p = input_path(entry)
        if p.suffix.lower() not in ('.c', '.cc', '.cpp', '.cxx') or not p.is_relative_to(w):
            continue
        rel = p.relative_to(w)
        overlay = p.parent == build/'native-httpd-rfc6455' and p.name in ('httpd_ws.c', 'httpd_parse.c')
        if not overlay and any(x.startswith('build') or x.startswith('.') for x in rel.parts[:-1]):
            continue
        result.append(entry)
    require(result, 'No actual project lint units')
    return result


def sdk_frontends(entries, build, dep, backend, donor):
    result = []
    for name in ('esp_http_client.c', 'esp_tls_mbedtls.c', 'ssl_tls.c'):
        selected = [e for e in entries if input_path(e).name == name]
        require(len(selected) == 1, 'Unique actual SDK/frontend input required: '+name)
        result.extend(selected)
    if not donor:
        require(input_path(result[0]).parent == build/'native-http-request-buffer', 'Actual patched HTTP-client source missing')
        require(input_path(result[1]) == (dep/'v6.0.2/components/esp-tls/esp_tls_mbedtls.c').resolve(), 'TLS EOF compile-definition source identity changed')
        require('-Dmbedtls_ssl_read=yoradio_mbedtls_ssl_read' in entry_tokens(result[1]), 'Actual TLS EOF source redirection missing')
        require(input_path(result[2]) == (dep/'v6.0.2/components/mbedtls/mbedtls/library/ssl_tls.c').resolve(), 'Default-OFF RX must use unmodified SDK SSL source')
        if backend == 'legacy':
            legacy = [e for e in entries if input_path(e).name == 'i2s_legacy.c']
            require(len(legacy) == 1 and input_path(legacy[0]) == (dep/'v5.5.4/components/driver/deprecated/i2s_legacy.c').resolve(), 'Wrong borrowed legacy driver')
            result.extend(legacy)
    return result


def delta(before, after):
    return {k: dict(before_present=k in before, before=before.get(k), after_present=k in after, after=after.get(k))
            for k in sorted(before.keys() | after.keys()) if k not in before or k not in after or
            type(before[k]) is not type(after[k]) or before[k] != after[k]}

def catalog(run, directory):
    result = {}
    diagnostic = re.compile(r'^(.+?):(\d+)(?::(\d+))?: (warning|error|fatal error): (.+)$')
    logs = set(run.rglob('*.log'))
    for path in run.glob('gcc-*/record.json'):
        for binding in read(path).get('native_build_logs', []):
            same(binding); logs.add(Path(binding['path']))
    for log in sorted(logs):
        if directory in log.parents:
            continue
        for number, line in enumerate(log.read_text(encoding='utf-8', errors='replace').splitlines(), 1):
            match = diagnostic.match(line)
            if match:
                require(match[4] == 'warning', 'Compiler/parser error remains in an actual phase log')
            if match or re.search(r'\bwarning:\s', line):
                key = hashlib.sha256(line.encode()).hexdigest()
                result.setdefault(key, dict(id=key, diagnostic=line, observations=[]))['observations'].append(dict(log=ident(log), line=number))
    value = dict(warning_free=not result, items=list(result.values()), acceptance_requires_exact_individual_root_triage=True)
    write(directory/'warnings.json', value)
    return value

    require(triage.get('warnings_sha256') == warning_binding['sha256'] and triage.get('source_commit') == r['final_commit'], 'Warning triage is not bound to actual final corpus')
    require({x['id'] for x in triage['items']} == {x['id'] for x in warnings['items']}, 'Every warning needs its own explicit triage')
    require(len(triage['items']) == len(warnings['items']), 'Duplicate warning dispositions')
    require(all(x.get('disposition') in ('accepted-existing', 'accepted-scoped') and x.get('reason') and x.get('reference') for x in triage['items']), 'Unexplained warning disposition')
    record.update(passed=True, source_release_id=r['source_release_id'], source_snapshot=source_check(r, w),
                  reports=closed, six_gcc_exports=True, six_actual_full_clang_profiles=True,
                  own_sdk_c3_object_lint_host_regression=True, warning_triage=ident(ARGS.triage),
                  ordinary_flac_watchdog_successor_included=bool(r.get('ordinary_flac_watchdog_evidence_files')),
                  physical_device_qualified=False, publication_authorized=False,
                  raw_privacy_inspection_still_required_before_any_staging_or_publication=True)


def main():
    global ARGS
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('phase', choices=['create', 'freeze', 'wrapper-test', 'gcc', 'clang-configure', 'clang-check',
                        'c3-gcc', 'c3-clang-configure', 'c3-clang-check', 'c3-host', 'warning-catalog', 'finalize'])
    parser.add_argument('--release', required=True, type=Path)
    parser.add_argument('--row', choices=[x[0] for x in ROWS])
    parser.add_argument('--delta-approval', type=Path)
    parser.add_argument('--triage', type=Path)
    ARGS = parser.parse_args()
    require(os.name == 'nt', 'Windows-native pinned orchestration required')
    r, w, dep, run = context(ARGS)
    name = ARGS.phase + ('-'+ARGS.row if ARGS.row else '')
    directory = run/name
    directory.mkdir(parents=True, exist_ok=False)
    record = dict(schema='cyd-final-production-phase-v1', phase=name, state='STARTING',
                  driver_pid=os.getpid(), started_epoch=time.time(), children=[], release=ident(ARGS.release),
                  free_before=shutil.disk_usage(REPO).free)
    write(directory/'record.json', record)
    try:
        if ARGS.phase not in ('create', 'freeze'):
            record['marker'] = ident(run/'marker.json'); checked_marker(r, w, dep, run)
        if ARGS.phase not in ('create', 'freeze', 'wrapper-test'):
            require(read(run/'wrapper-test/record.json')['state'] == 'TERMINAL_SUCCESS', 'Host shim and canonical wrapper gates must pass before later phases')
        require(shutil.disk_usage(REPO).free >= 1024**3, 'Operational disk reserve exhausted; no automatic cleanup')
        if ARGS.phase == 'create':
            create(r, w, dep, run, directory, record)
        elif ARGS.phase == 'freeze':
            freeze(r, w, dep, run, directory, record)
        elif ARGS.phase == 'wrapper-test':
            shim_host(r, w, dep, directory, record)
            process([dep/'python-3.12.10/tools/python.exe', '-B', w/'tests/run-cyd-production-wrapper.py', '--output', directory/'actual-host'], directory, 'wrapper-tests', record, w)
            require(read(directory/'actual-host/report.json')['status'] == 'PASS', 'Canonical wrapper host qualification failed')
        elif ARGS.phase in ('gcc', 'clang-configure', 'clang-check'):
            require(ARGS.row, 'Explicit one-row invocation required')
            row, backend, custom = next(x for x in ROWS if x[0] == ARGS.row)
            if ARGS.phase == 'gcc':
                gcc(r, w, dep, run, directory, record, row, backend, custom)
            elif ARGS.phase == 'clang-configure':
                clang_configure(r, w, dep, run, directory, record, row, backend, custom)
            else:
                require(ARGS.delta_approval, 'Explicit full typed-delta approval file required')
                clang_check(r, w, dep, run, directory, record, row, backend, custom)
        elif ARGS.phase == 'c3-gcc':
            donor_gcc(r, w, dep, run, directory, record)
        elif ARGS.phase == 'c3-clang-configure':
            clang_configure(r, w, dep, run, directory, record, '', None, False, True)
        elif ARGS.phase == 'c3-clang-check':
            require(ARGS.delta_approval, 'Own C3 typed-delta approval required')
            clang_check(r, w, dep, run, directory, record, '', None, False, True)
        elif ARGS.phase == 'c3-host':
            host(r, w, dep, run, directory, record)
        elif ARGS.phase == 'warning-catalog':
            record['warnings'] = catalog(run, directory)
        elif ARGS.phase == 'finalize':
            require(ARGS.triage, 'Exact per-warning root triage file required')
            finalize(r, w, dep, run, directory, record)
        if ARGS.phase not in ('create', 'freeze'):
            checked_marker(r, w, dep, run)
        record.update(state='TERMINAL_SUCCESS', ended_epoch=time.time(), free_after=shutil.disk_usage(REPO).free)
    except BaseException as error:
        record.update(state='TERMINAL_FAILURE', error_type=type(error).__name__, error=str(error), ended_epoch=time.time())
        raise
    finally:
        write(directory/'record.json', record)
    print(json.dumps(dict(phase=name, state=record['state'], record=ident(directory/'record.json'))), flush=True)


if __name__ == '__main__':
    main()
