def need(value,reason):

def identity(path):
    body=Path(path).read_bytes();return dict(bytes=len(body),sha256=hashlib.sha256(body).hexdigest())


def sha(path):return identity(path)['sha256']


def safe_path(path):

def contained(base,relative):

    need(not path.is_absolute() and '..' not in path.parts,'contained relative evidence path')
    result=(base/path).resolve();need(result.is_relative_to(base.resolve()),'no evidence symlink escape')
    return result


def check_pin(actual,expected,label):

def query_result(operation):
    try:return dict(state='observed',value=operation())
    except Exception as error:return dict(state='unknown',error_type=type(error).__name__,winerror=getattr(error,'winerror',None))


class PassiveWindowsSampler:
    """Queries only an owned process handle and its threads. Never sets policy."""
    def __init__(self,pid,handle):
        self.pid,self.handle=pid,wintypes.HANDLE(int(handle))
        self.kernel=ctypes.WinDLL('kernel32',use_last_error=True)
        self.kernel.GetProcessTimes.argtypes=[wintypes.HANDLE,*([ctypes.c_void_p]*4)]
        self.kernel.GetThreadTimes.argtypes=self.kernel.GetProcessTimes.argtypes
        self.kernel.GetPriorityClass.argtypes=[wintypes.HANDLE];self.kernel.GetPriorityClass.restype=wintypes.DWORD
        self.kernel.GetProcessAffinityMask.argtypes=[wintypes.HANDLE,ctypes.c_void_p,ctypes.c_void_p]
        self.kernel.GetProcessInformation.argtypes=[wintypes.HANDLE,ctypes.c_int,ctypes.c_void_p,wintypes.DWORD]
        self.kernel.CreateToolhelp32Snapshot.argtypes=[wintypes.DWORD,wintypes.DWORD];self.kernel.CreateToolhelp32Snapshot.restype=wintypes.HANDLE
        self.kernel.Thread32First.argtypes=[wintypes.HANDLE,ctypes.c_void_p];self.kernel.Thread32Next.argtypes=self.kernel.Thread32First.argtypes
        self.kernel.OpenThread.argtypes=[wintypes.DWORD,wintypes.BOOL,wintypes.DWORD];self.kernel.OpenThread.restype=wintypes.HANDLE
        self.kernel.GetProcessIdOfThread.argtypes=[wintypes.HANDLE];self.kernel.GetProcessIdOfThread.restype=wintypes.DWORD
        self.kernel.CloseHandle.argtypes=[wintypes.HANDLE]
        self.kernel.QueryPerformanceCounter.argtypes=[ctypes.POINTER(ctypes.c_longlong)]
        self.kernel.QueryPerformanceFrequency.argtypes=[ctypes.POINTER(ctypes.c_longlong)]

    def checked(self,ok):
        if not ok:raise ctypes.WinError(ctypes.get_last_error())

    def cpu(self,handle,thread=False):
        values=[ctypes.c_ulonglong() for _ in range(4)]
        self.checked((self.kernel.GetThreadTimes if thread else self.kernel.GetProcessTimes)(handle,*[ctypes.byref(v) for v in values]))
        return dict(created_100ns=values[0].value,kernel_100ns=values[2].value,user_100ns=values[3].value)

    def power(self):
        class State(ctypes.Structure):_fields_=[('Version',wintypes.ULONG),('ControlMask',wintypes.ULONG),('StateMask',wintypes.ULONG)]
        state=State(1,0,0);self.checked(self.kernel.GetProcessInformation(self.handle,4,ctypes.byref(state),ctypes.sizeof(state)))
        return dict(version=state.Version,control_mask=state.ControlMask,state_mask=state.StateMask)

    def affinity(self):
        process,system=ctypes.c_size_t(),ctypes.c_size_t()
        self.checked(self.kernel.GetProcessAffinityMask(self.handle,ctypes.byref(process),ctypes.byref(system)))
        return dict(process_mask=process.value,system_mask=system.value)

    def priority(self):
        value=self.kernel.GetPriorityClass(self.handle);self.checked(value);return int(value)

    def threads(self):
        class Entry(ctypes.Structure):
            _fields_=[('dwSize',wintypes.DWORD),('cntUsage',wintypes.DWORD),('thread',wintypes.DWORD),
                ('owner',wintypes.DWORD),('base_priority',wintypes.LONG),('delta_priority',wintypes.LONG),('flags',wintypes.DWORD)]
        snapshot=self.kernel.CreateToolhelp32Snapshot(4,0);self.checked(snapshot not in (None,ctypes.c_void_p(-1).value))
        result=[]
        try:
            entry=Entry();entry.dwSize=ctypes.sizeof(entry);more=self.kernel.Thread32First(snapshot,ctypes.byref(entry))
            self.checked(more)
            while more:
                if entry.owner==self.pid:
                    handle=self.kernel.OpenThread(0x0800,False,entry.thread)
                    if handle:
                        try:
                            def owned_cpu():
                                owner=self.kernel.GetProcessIdOfThread(handle);self.checked(owner)
                                need(owner==self.pid,'thread handle still belongs to owned child')
                                return self.cpu(handle,True)
                            result.append(dict(thread_id=entry.thread,base_priority=entry.base_priority,cpu=query_result(owned_cpu)))
                        finally:self.kernel.CloseHandle(handle)
                    else:result.append(dict(thread_id=entry.thread,cpu=dict(state='unknown',winerror=ctypes.get_last_error())))
                more=self.kernel.Thread32Next(snapshot,ctypes.byref(entry))
            need(ctypes.get_last_error()==18,'thread snapshot ended normally')
            return result
        finally:self.kernel.CloseHandle(snapshot)

    def qpc(self):
        counter,frequency=ctypes.c_longlong(),ctypes.c_longlong()
        self.checked(self.kernel.QueryPerformanceCounter(ctypes.byref(counter)))
        self.checked(self.kernel.QueryPerformanceFrequency(ctypes.byref(frequency)))
        return dict(counter=counter.value,frequency=frequency.value)

    def sample(self):
        return dict(pid=self.pid,wall_time_ns=time.time_ns(),monotonic_ns=time.monotonic_ns(),
            qpc=query_result(self.qpc),cpu=query_result(lambda:self.cpu(self.handle)),threads=query_result(self.threads),
            priority=query_result(self.priority),affinity=query_result(self.affinity),power_throttling=query_result(self.power),
            effective_timer_resolution_ms=None,timeBeginPeriod_return_unobserved=True,policy_changes=False)
